```dart
import 'package:equatable/equatable.dart';
import 'dart:convert';
import 'player.dart';
import 'role.dart';

enum GamePhase { waiting, night, day, voting, ended }

class Game extends Equatable {
  final String id;
  final String name;
  final DateTime createdAt;
  final DateTime? endedAt;
  final List<Player> players;
  final List<Role> availableRoles;
  final int currentRound;
  final GamePhase phase;
  final Player? winner;
  final List<String> actionLog;

  const Game({
    required this.id,
    required this.name,
    required this.createdAt,
    this.endedAt,
    this.players = const [],
    this.availableRoles = const [],
    this.currentRound = 0,
    this.phase = GamePhase.waiting,
    this.winner,
    this.actionLog = const [],
  });

  Game copyWith({
    String? id,
    String? name,
    DateTime? createdAt,
    DateTime? endedAt,
    List<Player>? players,
    List<Role>? availableRoles,
    int? currentRound,
    GamePhase? phase,
    Player? winner,
    List<String>? actionLog,
  }) {
    return Game(
      id: id ?? this.id,
      name: name ?? this.name,
      createdAt: createdAt ?? this.createdAt,
      endedAt: endedAt ?? this.endedAt,
      players: players ?? this.players,
      availableRoles: availableRoles ?? this.availableRoles,
      currentRound: currentRound ?? this.currentRound,
      phase: phase ?? this.phase,
      winner: winner ?? this.winner,
      actionLog: actionLog ?? this.actionLog,
    );
  }

  factory Game.fromMap(Map<String, dynamic> map) {
    final rolesJson = jsonDecode(map['roles_json']) as List;
    final playersJson = jsonDecode(map['players_json']) as List;
    final logJson = jsonDecode(map['log_json']) as List;

    return Game(
      id: map['id'],
      name: map['name'],
      createdAt: DateTime.fromMillisecondsSinceEpoch(map['created_at']),
      endedAt: map['ended_at'] != null
          ? DateTime.fromMillisecondsSinceEpoch(map['ended_at'])
          : null,
      players: playersJson.map((p) => Player.fromJson(p)).toList(),
      availableRoles: rolesJson.map((r) => Role.fromTypeString(r)).toList(),
      currentRound: map['current_round'],
      phase: GamePhase.values.firstWhere((e) => e.name == map['phase']),
      winner: map['winner'].isNotEmpty ? Player(name: map['winner'], id: '', colorHex: '') : null,
      actionLog: logJson.map((e) => e.toString()).toList(),
    );
  }

  Map<String, dynamic> toMap() {
    return {
      'id': id,
      'name': name,
      'created_at': createdAt.millisecondsSinceEpoch,
      'ended_at': endedAt?.millisecondsSinceEpoch,
      'players_json': jsonEncode(players.map((p) => p.toJson()).toList()),
      'roles_json': jsonEncode(availableRoles.map((r) => r.type.name).toList()),
      'current_round': currentRound,
      'phase': phase.name,
      'winner': winner?.name ?? '',
      'log_json': jsonEncode(actionLog),
    };
  }

  @override
  List<Object?> get props => [id, name, phase, currentRound];
}
```

## 10. models/game_state.dart

```dart
import 'package:equatable/equatable.dart';
import 'package:collection/collection.dart';
import 'game.dart';
import 'player.dart';
import 'role.dart';

enum GamePhase { night, day, voting, ended }

class GameState extends Equatable {
  final Game game;
  final GamePhase phase;
  final int round;
  final int timeLeft;
  final Map<String, String?> nightActions; // playerId -> targetId
  final Map<String, int> votes; // candidateId -> count
  final List<String> votingResults; // list of candidateIds
  final List<String> eliminationCandidates; // for second tour

  const GameState({
    required this.game,
    this.phase = GamePhase.night,
    this.round = 0,
    this.timeLeft = 0,
    this.nightActions = const {},
    this.votes = const {},
    this.votingResults = const [],
    this.eliminationCandidates = const [],
  });

  List<Player> get alivePlayers => game.players.where((p) => p.isAlive).toList();
  List<Player> get nightActivePlayers =>
      game.players.where((p) => p.isAlive && p.role != null && p.role!.isActiveAtNight).toList();
  List<Player> get mafiaPlayers =>
      game.players.where((p) => p.isAlive && p.role != null && (p.role!.type == RoleType.mafia || p.role!.type == RoleType.don)).toList();
  Player? get doctor => game.players.firstWhereOrNull((p) => p.isAlive && p.role?.type == RoleType.doctor);
  Player? get commissioner => game.players.firstWhereOrNull((p) => p.isAlive && p.role?.type == RoleType.commissioner);
  Player? get prostitute => game.players.firstWhereOrNull((p) => p.isAlive && p.role?.type == RoleType.prostitute);

  bool get mafiaWon {
    final aliveMafia = mafiaPlayers.length;
    final aliveCivilians = game.players.where((p) => p.isAlive && p.role != null && p.role!.type != RoleType.mafia && p.role!.type != RoleType.don).length;
    return aliveMafia >= aliveCivilians;
  }

  bool get civiliansWon => mafiaPlayers.isEmpty;

  GameState copyWith({
    Game? game,
    GamePhase? phase,
    int? round,
    int? timeLeft,
    Map<String, String?>? nightActions,
    Map<String, int>? votes,
    List<String>? votingResults,
    List<String>? eliminationCandidates,
  }) {
    return GameState(
      game: game ?? this.game,
      phase: phase ?? this.phase,
      round: round ?? this.round,
      timeLeft: timeLeft ?? this.timeLeft,
      nightActions: nightActions ?? this.nightActions,
      votes: votes ?? this.votes,
      votingResults: votingResults ?? this.votingResults,
      eliminationCandidates: eliminationCandidates ?? this.eliminationCandidates,
    );
  }

  @override
  List<Object?> get props => [phase, round, timeLeft, nightActions.length, votes.length];
}
```

```dart
import 'package:sqflite/sqflite.dart';
import 'package:path/path.dart';
import '../models/game.dart';

class DatabaseService {
  static final DatabaseService _instance = DatabaseService._internal();
  factory DatabaseService() => _instance;
  DatabaseService._internal();

  Database? _database;

  Future<Database> get database async {
    if (_database != null) return _database!;
    _database = await _initDB();
    return _database!;
  }

  Future<Database> _initDB() async {
    String path = join(await getDatabasesPath(), 'mafia_game.db');
    return await openDatabase(
      path,
      version: 1,
      onCreate: (db, version) async {
        await db.execute('''
          CREATE TABLE games (
            id TEXT PRIMARY KEY,
            name TEXT,
            created_at INTEGER,
            ended_at INTEGER,
            players_json TEXT,
            roles_json TEXT,
            current_round INTEGER,
            phase TEXT,
            winner TEXT,
            log_json TEXT
          )
        ''');
      },
    );
  }

  Future<void> insertGame(Game game) async {
    final db = await database;
    await db.insert('games', game.toMap(), conflictAlgorithm: ConflictAlgorithm.replace);
  }

  Future<void> updateGame(Game game) async {
    final db = await database;
    await db.update(
      'games',
      game.toMap(),
      where: 'id = ?',
      whereArgs: [game.id],
    );
  }

  Future<Game?> getGame(String id) async {
    final db = await database;
    final List<Map<String, dynamic>> maps = await db.query(
      'games',
      where: 'id = ?',
      whereArgs: [id],
    );
    if (maps.isNotEmpty) {
      return Game.fromMap(maps.first);
    }
    return null;
  }

  Future<List<Game>> getAllGames() async {
    final db = await database;
    final maps = await db.query('games', orderBy: 'created_at DESC');
    return maps.map((m) => Game.fromMap(m)).toList();
  }

  Future<void> deleteGame(String id) async {
    final db = await database;
    await db.delete('games', where: 'id = ?', whereArgs: [id]);
  }
}
```

## 12. services/ble_service.dart

```dart
import 'dart:convert';
import 'dart:typed_data';
import 'package:flutter_blue_plus/flutter_blue_plus.dart';
import 'package:riverpod/riverpod.dart';
import '../core/constants.dart';
import '../models/player.dart';

final bleServiceProvider = Provider<BleService>((ref) => BleService(ref));

class BleService {
  final Ref ref;
  final FlutterBluePlus _ble = FlutterBluePlus();
  final Map<String, BluetoothDevice> _playerDeviceMap = {};
  final Map<String, BluetoothCharacteristic> _txCharMap = {};
  final Map<String, BluetoothCharacteristic> _rxCharMap = {};
  StreamController<Map<String, dynamic>> _messageController = StreamController.broadcast();

  BleService(this.ref) {
    _initListeners();
  }

  void _initListeners() {
    _ble.onCharacteristicReceived.listen((event) {
      final device = event.device;
      final data = utf8.decode(event.data);
      try {
        final message = jsonDecode(data) as Map<String, dynamic>;
        // Добавляем отправителя
        message['_deviceId'] = device.remoteId;
        _messageController.add(message);
      } catch (e) {
        print('BLE parse error: $e');
      }
    });
  }

  Stream<Map<String, dynamic>> get onMessage => _messageController.stream;

  // ---- Ведущий ----
  Future<void> startHosting(String gameCode) async {
    final service = BluetoothService(
      uuid: SERVICE_UUID,
      characteristics: [
        BluetoothCharacteristic(
          uuid: RX_UUID,
          properties: CharacteristicProperties.writeWithoutResponse,
          permissions: CharacteristicPermissions.write,
        ),
        BluetoothCharacteristic(
          uuid: TX_UUID,
          properties: CharacteristicProperties.notify,
          permissions: CharacteristicPermissions.read,
        ),
      ],
    );

    await _ble.startAdvertising(
      advertisementData: AdvertisementData(
        localName: 'Mafia_$gameCode',
        serviceUuids: [service.uuid],
      ),
      services: [service],
    );
  }

  Future<void> stopHosting() async {
    await _ble.stopAdvertising();
  }

  // ---- Игрок ----
  Future<void> scanAndConnect(String gameCode, String playerName, String colorHex) async {
    final scanResult = await _ble.scan(
      withServices: [],
      timeout: Duration(seconds: 10),
    ).where((result) => result.device.localName == 'Mafia_$gameCode').first;
    final device = scanResult.device;
    await device.connect(timeout: Duration(seconds: 10));
    await device.discoverServices();
    final services = device.servicesList;
    BluetoothCharacteristic? rx, tx;
    for (var s in services) {
      for (var c in s.characteristics) {
        if (c.uuid == RX_UUID) rx = c;
        if (c.uuid == TX_UUID) tx = c;
      }
    }
    if (rx == null || tx == null) throw Exception('BLE характеристики не найдены');
    await tx.setNotifyValue(true);
    _rxCharMap[device.remoteId] = rx;
    _txCharMap[device.remoteId] = tx;
    // Отправляем регистрацию
    await sendToHost(device, {
      'type': 'register',
      'name': playerName,
      'color': colorHex,
    });
  }

  Future<void> disconnect() async {
    for (var device in _playerDeviceMap.values) {
      await device.disconnect();
    }
    _playerDeviceMap.clear();
    _txCharMap.clear();
    _rxCharMap.clear();
  }

  // ---- Отправка сообщений ----
  Future<void> sendToHost(BluetoothDevice device, Map<String, dynamic> message) async {
    final rx = _rxCharMap[device.remoteId];
    if (rx == null) throw Exception('RX характеристика не найдена');
    await _writeChar(rx, message);
  }

  Future<void> sendToPlayer(String playerId, Map<String, dynamic> message) async {
    final device = _playerDeviceMap[playerId];
    if (device == null) throw Exception('Устройство не найдено для игрока $playerId');
    final tx = _txCharMap[device.remoteId];
    if (tx == null) throw Exception('TX характеристика не найдена');
    await _writeChar(tx, message);
  }

  Future<void> broadcastToAll(Map<String, dynamic> message) async {
    for (var entry in _playerDeviceMap.entries) {
      await sendToPlayer(entry.key, message);
    }
  }

  Future<void> _writeChar(BluetoothCharacteristic char, Map<String, dynamic> message) async {
    final data = utf8.encode(jsonEncode(message));
    const int chunkSize = 20;
    for (int i = 0; i < data.length; i += chunkSize) {
      final end = i + chunkSize > data.length ? data.length : i + chunkSize;
      final chunk = data.sublist(i, end);
      await char.write(Uint8List.fromList(chunk), withoutResponse: true);
    }
  }

  // ---- Управление подключениями ----
  void registerPlayerDevice(String playerId, BluetoothDevice device) {
    _playerDeviceMap[playerId] = device;
    _monitorConnection(device, playerId);
  }

  void _monitorConnection(BluetoothDevice device, String playerId) {
    device.connectionState.listen((state) {
      if (state == BluetoothConnectionState.disconnected) {
        _attemptReconnect(playerId, device);
      }
    });
  }

  Future<void> _attemptReconnect(String playerId, BluetoothDevice device) async {
    int attempts = 0;
    while (attempts < 3) {
      attempts++;
      try {
        await device.connect(timeout: Duration(seconds: 5));
        // Успешно переподключились
        ref.read(gameControllerProvider.notifier).onPlayerReconnected(playerId);
        return;
      } catch (e) {
        await Future.delayed(Duration(seconds: 2));
      }
    }
    ref.read(gameControllerProvider.notifier).onPlayerDisconnected(playerId);
  }

  void clearDevices() {
    _playerDeviceMap.clear();
    _txCharMap.clear();
    _rxCharMap.clear();
  }

  BluetoothDevice? getDeviceForPlayer(String playerId) => _playerDeviceMap[playerId];
}
```

```dart
class GameSetupState {
  final int playerCount;
  final Map<RoleType, int> roleCounts;
  final String gameName;

  GameSetupState({
    this.playerCount = 10,
    this.roleCounts = const {},
    this.gameName = '',
  });

  GameSetupState copyWith({
    int? playerCount,
    Map<RoleType, int>? roleCounts,
    String? gameName,
  }) {
    return GameSetupState(
      playerCount: playerCount ?? this.playerCount,
      roleCounts: roleCounts ?? this.roleCounts,
      gameName: gameName ?? this.gameName,
    );
  }
}

class GameSetupController extends StateNotifier<GameSetupState> {
  GameSetupController() : super(GameSetupState()) {
    _applyPreset(10);
  }

  void setPlayerCount(int count) {
    state = state.copyWith(playerCount: count);
    _applyPreset(count);
  }

  void setGameName(String name) {
    state = state.copyWith(gameName: name);
  }

  void incrementRole(RoleType role) {
    final newCount = (state.roleCounts[role] ?? 0) + 1;
    if (_totalRoles() + 1 > state.playerCount) return;
    final newMap = Map<RoleType, int>.from(state.roleCounts)..[role] = newCount;
    state = state.copyWith(roleCounts: newMap);
    _balanceCivilian();
  }

  void decrementRole(RoleType role) {
    final current = state.roleCounts[role] ?? 0;
    if (current == 0) return;
    final newMap = Map<RoleType, int>.from(state.roleCounts)..[role] = current - 1;
    state = state.copyWith(roleCounts: newMap);
    _balanceCivilian();
  }

  int _totalRoles() => state.roleCounts.values.fold(0, (a,b) => a+b);

  void _balanceCivilian() {
    final total = _totalRoles();
    final civilianCount = state.playerCount - total;
    if (civilianCount >= 0) {
      final newMap = Map<RoleType, int>.from(state.roleCounts)..[RoleType.civilian] = civilianCount;
      state = state.copyWith(roleCounts: newMap);
    }
  }

  void _applyPreset(int count) {
    Map<RoleType, int> preset = {RoleType.civilian: count};
    if (count >= 6) {
      preset[RoleType.mafia] = 2;
      preset[RoleType.doctor] = 1;
      preset[RoleType.commissioner] = 1;
    }
    if (count >= 9) {
      preset[RoleType.mafia] = 3;
      preset[RoleType.don] = 1;
      preset[RoleType.prostitute] = 1;
    }
    if (count >= 13) {
      preset[RoleType.mafia] = 4;
      preset[RoleType.maniac] = 1;
    }
    if (count >= 17) {
      preset[RoleType.mafia] = 5;
      preset[RoleType.sheriff] = 1;
    }
    // Пересчёт мирных
    final totalRoles = preset.values.fold(0, (a,b) => a+b);
    preset[RoleType.civilian] = count - (totalRoles - (preset[RoleType.civilian] ?? 0));
    state = state.copyWith(roleCounts: preset);
  }

  Future<Game> createGame() async {
    if (_totalRoles() != state.playerCount) {
      throw Exception('Количество ролей не совпадает с числом игроков');
    }
    final code = generateGameCode();
    final roles = <Role>[];
    state.roleCounts.forEach((roleType, count) {
      for (int i = 0; i < count; i++) {
        roles.add(Role.fromType(roleType));
      }
    });
    final game = Game(
      id: code,
      name: state.gameName.isEmpty ? 'Игра от ${DateTime.now().toLocal().toString().substring(0, 16)}' : state.gameName,
      createdAt: DateTime.now(),
      availableRoles: roles,
    );
    await DatabaseService().insertGame(game);
    return game;
  }
}
```

## 18. features/host_game/game_controller.dart

```dart
import 'dart:async';
import 'package:riverpod/riverpod.dart';
import '../../models/game.dart';
import '../../models/game_state.dart';
import '../../models/player.dart';
import '../../services/ble_service.dart';
import '../../services/database_service.dart';
import '../../services/audio_service.dart';
import '../../providers/game_providers.dart';

final gameControllerProvider = StateNotifierProvider<GameController, GameState>((ref) {
  return GameController(ref);
});

class GameController extends StateNotifier<GameState> {
  final Ref ref;
  Timer? _phaseTimer;
  Timer? _actionTimer;
  bool _isPaused = false;

  GameController(this.ref) : super(GameState(game: ref.read(currentGameProvider)!));

  void initializeGame(Game game) {
    state = GameState(game: game, phase: GamePhase.night, round: 1);
    _saveGameState();
  }

  void startNight() {
    final audio = AudioService();
    audio.playSound(AudioService.nightSound);
    audio.playMusic('sounds/night_music.mp3');
    state = state.copyWith(phase: GamePhase.night, timeLeft: _calculateNightDuration());
    _startPhaseTimer();
    _startActionCollection();
    _broadcastPhaseUpdate();
    _addLog('Начинается ночь ${state.round}');
  }

  void startDay() {
    final audio = AudioService();
    audio.playSound(AudioService.daySound);
    audio.stopMusic();
    _resolveNightActions();
    state = state.copyWith(phase: GamePhase.day, timeLeft: _calculateDayDuration());
    _startPhaseTimer();
    _broadcastPhaseUpdate();
    _addLog('Наступает день ${state.round}');
  }

  void startVoting() {
    final audio = AudioService();
    audio.playSound(AudioService.voteSound);
    state = state.copyWith(phase: GamePhase.voting, timeLeft: 30);
    _startPhaseTimer();
    state.votes.clear();
    state.votingResults.clear();
    state.eliminationCandidates.clear();
    _broadcastPhaseUpdate();
    _addLog('Начинается голосование');
  }

  void endGame(String winner) {
    final audio = AudioService();
    audio.playSound(AudioService.winSound);
    audio.stopMusic();
    state = state.copyWith(phase: GamePhase.ended);
    _phaseTimer?.cancel();
    _actionTimer?.cancel();
    _broadcastPhaseUpdate();
    _addLog('Игра окончена. Победила $winner');
    state.game.endedAt = DateTime.now();
    state.game.winner = Player(name: winner, id: '', colorHex: '');
    _saveGameState();
  }

  void togglePause() {
    _isPaused = !_isPaused;
    if (_isPaused) {
      _phaseTimer?.cancel();
      _phaseTimer = null;
    } else {
      _startPhaseTimer();
    }
    _broadcastPause(_isPaused);
  }

  void addTime(int seconds) {
    state = state.copyWith(timeLeft: state.timeLeft + seconds);
    _broadcastTimeUpdate();
  }

  int _calculateNightDuration() {
    final activeCount = state.nightActivePlayers.length;
    return 10 + activeCount * 2;
  }

  int _calculateDayDuration() {
    return state.alivePlayers.length * 60;
  }

  void _startPhaseTimer() {
    _phaseTimer?.cancel();
    _phaseTimer = Timer.periodic(Duration(seconds: 1), (timer) {
      if (_isPaused) return;
      if (state.timeLeft > 0) {
        state = state.copyWith(timeLeft: state.timeLeft - 1);
        _broadcastTimeUpdate();
      } else {
        timer.cancel();
        _onPhaseTimeout();
      }
    });
  }

  void _onPhaseTimeout() {
    switch (state.phase) {
      case GamePhase.night:
        _finishNight();
        break;
      case GamePhase.day:
        startVoting();
        break;
      case GamePhase.voting:
        _resolveVoting();
        break;
      case GamePhase.ended:
        break;
    }
  }

  void _startActionCollection() {
    state.nightActions.clear();
    final activePlayers = state.nightActivePlayers;
    for (var player in activePlayers) {
      _sendMessageToPlayer(player, {
        'type': 'night_action',
        'role': player.role!.type.name,
        'alive_players': state.alivePlayers.map((p) => {'id': p.id, 'name': p.name}).toList(),
        'timeout': 10,
      });
    }
    _actionTimer = Timer(Duration(seconds: _calculateNightDuration()), () {
      _finishNight();
    });
  }

  void onNightActionReceived(String playerId, String? targetId) {
    state.nightActions[playerId] = targetId;
    final allAnswered = state.nightActivePlayers.every((p) => state.nightActions.containsKey(p.id));
    if (allAnswered) {
      _actionTimer?.cancel();
      _finishNight();
    }
  }

  void _finishNight() {
    _actionTimer?.cancel();
    _resolveNightActions();
    startDay();
  }

  void _resolveNightActions() {
    final actions = state.nightActions;
    String? killedPlayer;
    String? healedPlayer;
    String? blockedPlayer;

    // Проститутка
    final prostitute = state.prostitute;
    if (prostitute != null && actions.containsKey(prostitute.id)) {
      final blockedId = actions[prostitute.id];
      if (blockedId != null) {
        blockedPlayer = state.game.players.firstWhere((p) => p.id == blockedId).name;
      }
    }

    // Доктор
    final doctor = state.doctor;
    if (doctor != null && actions.containsKey(doctor.id)) {
      final healedId = actions[doctor.id];
      if (healedId != null) {
        healedPlayer = state.game.players.firstWhere((p) => p.id == healedId).name;
      }
    }

    // Мафия
    String? killTarget;
    final mafiaIds = state.mafiaPlayers.map((p) => p.id).toSet();
    for (var id in mafiaIds) {
      if (actions.containsKey(id) && actions[id] != null) {
        killTarget = actions[id];
        break;
      }
    }
    final don = state.game.players.firstWhereOrNull((p) => p.isAlive && p.role?.type == RoleType.don);
    if (don != null && actions.containsKey(don.id) && actions[don.id] != null) {
      killTarget = actions[don.id];
    }

    if (killTarget != null && blockedPlayer != null &&
        state.game.players.firstWhere((p) => p.id == killTarget).name == blockedPlayer) {
      killTarget = null;
    }

    if (killTarget != null && (healedPlayer == null || healedPlayer != state.game.players.firstWhere((p) => p.id == killTarget).name)) {
      final killed = state.game.players.firstWhere((p) => p.id == killTarget);
      killed.isAlive = false;
      killedPlayer = killed.name;
      _addLog('Мафия убила $killedPlayer');
    } else if (killTarget != null) {
      _addLog('Мафия пыталась убить, но жертва вылечена');
    }

    // Комиссар
    final commissioner = state.commissioner;
    if (commissioner != null && actions.containsKey(commissioner.id)) {
      final checkedId = actions[commissioner.id];
      if (checkedId != null) {
        final checked = state.game.players.firstWhere((p) => p.id == checkedId);
        final isMafia = checked.role != null && (checked.role!.type == RoleType.mafia || checked.role!.type == RoleType.don);
        _addLog('Комиссар проверил ${checked.name}. ${isMafia ? 'Мафия!' : 'Мирный'}');
        _sendMessageToPlayer(commissioner, {
          'type': 'check_result',
          'target': checked.name,
          'is_mafia': isMafia,
        });
      }
    }

    _broadcastNightResult(killedPlayer);
    _saveGameState();
  }

  void onVoteReceived(String playerId, String candidateId) {
    final player = state.game.players.firstWhere((p) => p.id == playerId);
    if (!player.isAlive) return;
    if (state.votes.containsKey(playerId)) return;

    state.votes[playerId] = candidateId;
    state.votingResults.add(candidateId);

    if (state.votes.length >= state.alivePlayers.length) {
      _resolveVoting();
    }
  }

  void _resolveVoting() {
    final voteCount = <String, int>{};
    for (var candidateId in state.votingResults) {
      voteCount[candidateId] = (voteCount[candidateId] ?? 0) + 1;
    }

    // Проверка "против всех"
    final noneVotes = voteCount['none'] ?? 0;
    if (noneVotes > 0 && noneVotes >= state.alivePlayers.length / 2) {
      _addLog('Город решил никого не выгонять');
      state.round++;
      startNight();
      return;
    }

    // Найти максимум
    int maxVotes = 0;
    String? eliminatedId;
    for (var entry in voteCount.entries) {
      if (entry.value > maxVotes) {
        maxVotes = entry.value;
        eliminatedId = entry.key;
      }
    }

    final candidatesWithMax = voteCount.entries.where((e) => e.value == maxVotes).map((e) => e.key).toList();
    if (candidatesWithMax.length > 1) {
      // Ничья
      state.eliminationCandidates = candidatesWithMax;
      _addLog('Ничья! Второй тур между ${candidatesWithMax.map((id) => state.game.players.firstWhere((p) => p.id == id).name).join(', ')}');
      _broadcastSecondTour();
      state.votes.clear();
      state.votingResults.clear();
      _phaseTimer?.cancel();
      state = state.copyWith(timeLeft: 20);
      _startPhaseTimer();
      return;
    }

    if (eliminatedId != null && eliminatedId != 'none') {
      final eliminated = state.game.players.firstWhere((p) => p.id == eliminatedId);
      eliminated.isAlive = false;
      _addLog('Город прогнал ${eliminated.name} (голосов: $maxVotes)');
      _broadcastElimination(eliminated.name);
      _checkWinCondition();
    }

    state.round++;
    startNight();
  }

  void _checkWinCondition() {
    if (state.mafiaWon) {
      endGame('Мафия');
      _broadcastWin('Мафия');
    } else if (state.civiliansWon) {
      endGame('Город');
      _broadcastWin('Город');
    } else {
      _addLog('Игра продолжается. Мафия: ${state.mafiaPlayers.length}, Город: ${state.alivePlayers.length - state.mafiaPlayers.length}');
    }
  }

  // ---- BLE communication ----
  void _sendMessageToPlayer(Player player, Map<String, dynamic> message) {
    final ble = ref.read(bleServiceProvider);
    final device = ble.getDeviceForPlayer(player.id);
    if (device != null) {
      ble.sendToPlayer(player.id, message);
    }
  }

  void _broadcastPhaseUpdate() {
    ref.read(bleServiceProvider).broadcastToAll({
      'type': 'phase_update',
      'phase': state.phase.name,
      'round': state.round,
      'time_left': state.timeLeft,
      'alive': state.alivePlayers.map((p) => p.name).toList(),
    });
  }

  void _broadcastTimeUpdate() {
    ref.read(bleServiceProvider).broadcastToAll({
      'type': 'timer_update',
      'time_left': state.timeLeft,
    });
  }

  void _broadcastNightResult(String? killed) {
    ref.read(bleServiceProvider).broadcastToAll({
      'type': 'night_result',
      'killed': killed,
    });
  }

  void _broadcastSecondTour() {
    final candidates = state.eliminationCandidates.map((id) => state.game.players.firstWhere((p) => p.id == id).name).toList();
    ref.read(bleServiceProvider).broadcastToAll({
      'type': 'second_tour',
      'candidates': candidates,
    });
  }

  void _broadcastElimination(String name) {
    ref.read(bleServiceProvider).broadcastToAll({
      'type': 'elimination',
      'name': name,
    });
  }

  void _broadcastWin(String winner) {
    ref.read(bleServiceProvider).broadcastToAll({
      'type': 'game_over',
      'winner': winner,
    });
  }

  void _broadcastPause(bool paused) {
    ref.read(bleServiceProvider).broadcastToAll({
      'type': 'pause',
      'paused': paused,
    });
  }

  void _addLog(String message) {
    state.game.actionLog.add('Раунд ${state.round}: $message');
    _saveGameState();
  }

  void _saveGameState() {
    DatabaseService().updateGame(state.game);
  }

  void onPlayerReconnected(String playerId) {
    final player = state.game.players.firstWhere((p) => p.id == playerId);
    player.isConnected = true;
    _addLog('${player.name} переподключился');
    _sendFullStateToPlayer(player);
  }

  void onPlayerDisconnected(String playerId) {
    final player = state.game.players.firstWhere((p) => p.id == playerId);
    player.isConnected = false;
    _addLog('${player.name} отключился');
    if (state.phase == GamePhase.night && state.nightActions.containsKey(playerId)) {
      state.nightActions[playerId] = null;
    }
  }

  void _sendFullStateToPlayer(Player player) {
    _sendMessageToPlayer(player, {
```

```dart
  Widget build(BuildContext context) {
    return Scaffold(
```

```dart
      body: Padding(
        padding: const EdgeInsets.all(20.0),
        child: Column(
          children: [
            TextField(
              controller: _codeController,
              decoration: InputDecoration(labelText: 'Код игры (6 цифр)'),
              maxLength: 6,
            ),
            TextField(
              controller: _nameController,
              decoration: InputDecoration(labelText: 'Ваше имя'),
            ),
            SizedBox(height: 16),
            Row(
              children: [
                Text('Цвет:'),
                Expanded(
                  child: Wrap(
                    spacing: 8,
                    children: getPresetColors().map((colorHex) {
                      return InkWell(
                        onTap: () => setState(() => selectedColor = colorHex),
                        child: CircleAvatar(
                          backgroundColor: Color(int.parse(colorHex)),
                          radius: 16,
                          child: selectedColor == colorHex
                              ? Icon(Icons.check, color: Colors.white, size: 16)
                              : null,
                        ),
                      );
                    }).toList(),
                  ),
                ),
              ],
            ),
            SizedBox(height: 20),
            ElevatedButton(
              onPressed: _connect,
              child: Text('Подключиться'),
            ),
          ],
        ),
      ),
    );
  }

  Future<void> _connect() async {
    final code = _codeController.text.trim();
    final name = _nameController.text.trim();
    if (code.isEmpty || name.isEmpty) {
      ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text('Заполните все поля')));
      return;
    }
    final ble = ref.read(bleServiceProvider);
    try {
      await ble.scanAndConnect(code, name, selectedColor);
      // Создаём игрока (роль будет назначена позже)
      final player = Player(
        id: generatePlayerId(),
        name: name,
        colorHex: selectedColor,
      );
      Navigator.pushReplacementNamed(context, '/player_game', arguments: player);
    } catch (e) {
      ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text('Ошибка: $e')));
    }
  }
}
```

```dart
import 'package:flutter/material.dart';
import 'package:riverpod/riverpod.dart';
import '../../services/ble_service.dart';
import '../../models/player.dart';
import '../../models/role.dart';

class PlayerGameScreen extends ConsumerStatefulWidget {
  final Player player;
  const PlayerGameScreen({Key? key, required this.player}) : super(key: key);

  @override
  _PlayerGameScreenState createState() => _PlayerGameScreenState();
}

class _PlayerGameScreenState extends ConsumerState<PlayerGameScreen> {
  String? selectedTargetId;
  bool isNightAction = false;
  bool isVoting = false;
  bool hasVoted = false;
  String phaseText = 'Ожидание...';
  int timeLeft = 0;
  String? myRole;
  List<Map<String, String>> alivePlayers = [];

  @override
  void initState() {
    super.initState();
    final ble = ref.read(bleServiceProvider);
    ble.onMessage.listen((message) {
      _handleMessage(message);
    });
  }

  void _handleMessage(Map<String, dynamic> message) {
    switch (message['type']) {
      case 'role':
        setState(() {
          myRole = message['role'];
        });
        break;
      case 'phase_update':
        setState(() {
          phaseText = message['phase'];
          timeLeft = message['time_left'] ?? 0;
          alivePlayers = (message['alive'] as List).map((name) => {'id': name, 'name': name}).toList();
        });
        break;
      case 'timer_update':
        setState(() {
          timeLeft = message['time_left'] ?? 0;
        });
        break;
      case 'night_action':
        setState(() {
          isNightAction = true;
          isVoting = false;
          selectedTargetId = null;
          alivePlayers = (message['alive_players'] as List).map((p) => {'id': p['id'], 'name': p['name']}).toList();
        });
        break;
      case 'vote_request':
        setState(() {
          isVoting = true;
          isNightAction = false;
          hasVoted = false;
          selectedTargetId = null;
        });
        break;
      case 'second_tour':
        setState(() {
          isVoting = true;
          hasVoted = false;
          selectedTargetId = null;
          alivePlayers = (message['candidates'] as List).map((name) => {'id': name, 'name': name}).toList();
        });
        break;
      case 'night_result':
        _showNightResult(message);
        break;
      case 'check_result':
        _showCheckResult(message);
        break;
      case 'elimination':
        _showElimination(message['name']);
        break;
      case 'game_over':
        _showGameOver(message['winner']);
        break;
      case 'pause':
        if (message['paused'] == true) {
          ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text('Игра на паузе')));
        } else {
          ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text('Игра продолжается')));
        }
        break;
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: Text('Игра'),
        actions: [
          IconButton(
            icon: Icon(Icons.info_outline),
            onPressed: _showRoleInfo,
          ),
        ],
      ),
      body: Padding(
        padding: const EdgeInsets.all(16.0),
        child: Column(
          children: [
            Card(
              child: Padding(
                padding: const EdgeInsets.all(16.0),
                child: Column(
                  children: [
                    Text('Ваша роль', style: TextStyle(fontSize: 18)),
                    Text(
                      myRole ?? 'Ожидание...',
                      style: TextStyle(fontSize: 32, fontWeight: FontWeight.bold),
                    ),
                    SizedBox(height: 8),
                    Text('Фаза: $phaseText', style: TextStyle(fontSize: 16)),
                    Text('Времени осталось: ${timeLeft ~/ 60}:${(timeLeft % 60).toString().padLeft(2, '0')}'),
                  ],
                ),
              ),
            ),
            SizedBox(height: 16),
            Expanded(
              child: _buildActionArea(),
            ),
          ],
        ),
      ),
    );
  }

  Widget _buildActionArea() {
    if (isNightAction && myRole != null) {
      return _buildChoiceList('Выберите цель', onConfirm: (id) {
        _sendAction('night_action', id);
      });
    } else if (isVoting && !hasVoted) {
      return _buildChoiceList('Голосуйте за кандидата', onConfirm: (id) {
        _sendAction('vote', id);
      }, showNone: true);
    } else if (hasVoted) {
      return Center(child: Text('Вы уже проголосовали', style: TextStyle(fontSize: 18)));
    } else {
      return Center(child: Text('Ожидайте действий...', style: TextStyle(fontSize: 18)));
    }
  }

  Widget _buildChoiceList(String title, {required Function(String) onConfirm, bool showNone = false}) {
    final options = alivePlayers.map((p) {
      return {'id': p['id']!, 'name': p['name']!};
    }).toList();
    if (showNone) {
      options.add({'id': 'none', 'name': 'Никто (против всех)'});
    }
    return Column(
      children: [
        Text(title, style: TextStyle(fontSize: 20)),
        Expanded(
          child: ListView.builder(
            itemCount: options.length,
            itemBuilder: (ctx, idx) {
              final item = options[idx];
              return RadioListTile<String>(
                title: Text(item['name']!),
                value: item['id']!,
                groupValue: selectedTargetId,
                onChanged: (val) => setState(() => selectedTargetId = val),
              );
            },
          ),
        ),
        ElevatedButton(
          onPressed: selectedTargetId != null
              ? () {
                  onConfirm(selectedTargetId!);
                  setState(() {
                    isNightAction = false;
                    isVoting = false;
                    hasVoted = true;
                  });
                }
              : null,
          child: Text('Подтвердить'),
        ),
      ],
    );
  }

  void _sendAction(String type, String targetId) {
    final ble = ref.read(bleServiceProvider);
    ble.sendToHost({
      'type': type,
      'target': targetId,
    });
  }

  void _showRoleInfo() {
    showDialog(
      context: context,
      builder: (ctx) => AlertDialog(
        title: Text('Описание роли'),
        content: Text(_getRoleDescription()),
        actions: [
          TextButton(onPressed: () => Navigator.pop(ctx), child: Text('Закрыть')),
        ],
      ),
    );
  }

  String _getRoleDescription() {
    // Можно хранить полное описание в модели
    switch (myRole) {
      case 'mafia': return 'Каждую ночь убиваете одного игрока. Днём притворяетесь мирным.';
      case 'don': return 'Глава мафии, выбираете жертву, мафия голосует.';
      case 'doctor': return 'Каждую ночь лечите одного игрока, спасая от смерти.';
      case 'commissioner': return 'Каждую ночь проверяете одного игрока на мафию.';
      case 'prostitute': return 'Каждую ночь блокируете одного игрока (он пропускает ход).';
      case 'maniac': return 'Убиваете каждую ночь, но не мафия. Побеждаете, если останетесь один.';
      case 'sheriff': return 'Один раз за игру можете днём проверить игрока.';
      default: return 'Мирный житель, голосуете днём.';
    }
  }

  void _showNightResult(Map<String, dynamic> message) {
    showDialog(
      context: context,
      builder: (ctx) => AlertDialog(

        content: Text(message['killed'] != null ? 'Убит: ${message['killed']}' : 'Этой ночью никто не умер'),
        actions: [
          TextButton(onPressed: () => Navigator.pop(ctx), child: Text('Ок')),
        ],
      ),
    );
  }

  void _showCheckResult(Map<String, dynamic> message) {
    showDialog(
      context: context,
      builder: (ctx) => AlertDialog(
        title: Text('Результат проверки'),
        content: Text('${message['target']} — ${message['is_mafia'] ? 'Мафия!' : 'Мирный'}'),
        actions: [
          TextButton(onPressed: () => Navigator.pop(ctx), child: Text('Ок')),
        ],
      ),
    );
  }

  void _showElimination(String name) {
    ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text('$name выбыл из города')));

  void _showGameOver(String winner) {
```
